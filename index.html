Sub Generate_Due_Drawn_Statement() Dim ws As Worksheet Set ws = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count)) ws.Name = "Due_Drawn_Statement" ' स्क्रीन रिफ्रेश बंद करणे Application.ScreenUpdating = False ' ---- युझर इनपुट गोळा करणे (उदाहरणादाखल मॅन्युअली किंवा फॉर्मद्वारे) ---- Dim CurrentBasic As Double, EffectiveDate As Date, IncrementMonth As Integer Dim CurrentLevel As String, ProposedLevel As String ' टेस्ट इनपुट (हे तुमच्या यूजर फॉर्मशी लिंक होईल) CurrentBasic = 35400 ' मूळ पगार CurrentLevel = "S-13" ProposedLevel = "S-14" EffectiveDate = #1/1/2026# ' अंमलबजावणी दिनांक IncrementMonth = 7 ' ७ = १ जुलै, १ = १ जानेवारी (युझर ऑप्शननुसार) ' ---- तक्त्याचे हेडिंग्ज आणि डिझाईन ---- ws.Range("A1:M1").Merge ws.Range("A1").Value = "वरिष्ठ श्रेणी वेतननिश्चिती व ड्यू-ड्रॉन मासिक फरक तक्ता" ws.Range("A1").Font.Bold = True ws.Range("A1").Font.Size = 14 ws.Range("A1").HorizontalAlignment = xlCenter ' टेबल हेडर्स Dim Headers As Variant Headers = Array("महिना/वर्ष", "देय पे लेव्हल", "देय मूळ वेतन", "देय महागाई भत्ता", "देय एकूण (Due)", _ "ओढलेले पे लेव्हल", "ओढलेले मूळ वेतन", "ओढलेले महागाई भत्ता", "ओढलेले एकूण (Drawn)", _ "मासिक फरक (Basic)", "मासिक फरक (DA)", "एकूण मासिक फरक", "शेरा") Dim i As Integer For i = 0 To UBound(Headers) ws.Cells(3, i + 1).Value = Headers(i) ws.Cells(3, i + 1).Font.Bold = True ws.Cells(3, i + 1).Interior.Color = RGB(220, 230, 241) Next i ' ---- २५ महिन्यांचे टप्पे आणि वार्षिक वेतनवाढ कॅल्क्युलेशन ---- Dim RowIndex As Integer: RowIndex = 4 Dim MonthCounter As Integer Dim RunningDate As Date: RunningDate = EffectiveDate Dim DueBasic As Double: DueBasic = CurrentBasic Dim DrawnBasic As Double: DrawnBasic = CurrentBasic ' जर जुन्याच दराने ओढला असेल तर ' सुचवलेल्या पे लेव्हलमध्ये पुढील टप्पा (वेतन निश्चिती नियमानुसार ३% वाढ) ' (टीप: प्रत्यक्ष रचनेसाठी 'PAY SCALE.xlsx' मधील मॅट्रिक्सचा वापर करावा) DueBasic = Application.WorksheetFunction.RoundUp(DueBasic * 1.03, -2) For MonthCounter = 1 To 24 ' पुढील २४-२५ महिन्यांचे टप्पे जनरेट करणे ' ** युझरने निवडलेल्या वार्षिक वेतनवाढ तारखेनुसार बेसिक वाढवणे ** If Month(RunningDate) = IncrementMonth And MonthCounter > 1 Then ' देय आणि ओढलेल्या वेतनात ३% वार्षिक वेतनवाढ (राउंडेड टू नेक्स्ट १००) DueBasic = Application.WorksheetFunction.RoundUp(DueBasic * 1.03, -2) DrawnBasic = Application.WorksheetFunction.RoundUp(DrawnBasic * 1.03, -2) End If ' डेटा भरणे ws.Cells(RowIndex, 1).Value = Format(RunningDate, "mmmm-yyyy") ws.Cells(RowIndex, 2).Value = ProposedLevel ws.Cells(RowIndex, 3).Value = DueBasic ws.Cells(RowIndex, 4).Formula = "=C" & RowIndex & "*0.50" ' महागाई भत्ता उदा. ५०% (बदली नियमानुसार राहील) ws.Cells(RowIndex, 5).Formula = "=C" & RowIndex & "+D" & RowIndex ws.Cells(RowIndex, 6).Value = CurrentLevel ws.Cells(RowIndex, 7).Value = DrawnBasic ws.Cells(RowIndex, 8).Formula = "=G" & RowIndex & "*0.50" ws.Cells(RowIndex, 9).Formula = "=G" & RowIndex & "+H" & RowIndex ' फरक कॅल्क्युलेशन फॉर्म्युला ws.Cells(RowIndex, 10).Formula = "=C" & RowIndex & "-G" & RowIndex ws.Cells(RowIndex, 11).Formula = "=D" & RowIndex & "-H" & RowIndex ws.Cells(RowIndex, 12).Formula = "=E" & RowIndex & "-I" & RowIndex If Month(RunningDate) = IncrementMonth Then ws.Cells(RowIndex, 13).Value = "वार्षिक वेतनवाढ" ws.Cells(RowIndex, 13).Font.Color = RGB(0, 128, 0) End If ' पुढील महिन्याकडे जाणे RunningDate = DateAdd("m", 1, RunningDate) RowIndex = RowIndex + 1 Next MonthCounter ' एकूण (Total) रो जोडणे ws.Cells(RowIndex, 1).Value = "एकूण फरक (Total)" ws.Cells(RowIndex, 1).Font.Bold = True ws.Range(ws.Cells(RowIndex, 10), ws.Cells(RowIndex, 12)).Formula = "=SUM(J4:J" & RowIndex - 1 & ")" ws.Range(ws.Cells(RowIndex, 10), ws.Cells(RowIndex, 12)).Font.Bold = True ' फॉर्मेटिंग आणि बॉर्डर ws.Range("A3:M" & RowIndex).Borders.LineStyle = xlContinuous ws.Columns("A:M").AutoFit Application.ScreenUpdating = True MsgBox "ड्यू-ड्रॉन मासिक फरक तक्ता यशस्वीरित्या तयार झाला आहे!", vbInformation End Sub
